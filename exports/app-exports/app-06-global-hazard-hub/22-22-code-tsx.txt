import { useEffect, useState } from 'react';
import { fetchRealtimeData } from '@/lib/api';
import NavBar from '@/components/NavBar';
import Sidebar from '@/components/Sidebar';
import { Box, Grid } from '@mui/material';

export default function Dashboard() {
  const [env, setEnv] = useState<Record<string, any>>({});

  useEffect(() => {
    const interval = setInterval(async () => {
      const data = await fetchRealtimeData();      // NOAA + custom sensors
      setEnv(data);
    }, 10000); // every 10 s
    return () => clearInterval(interval);
  }, []);

  return (
    <>
      <NavBar />
      <Sidebar />
      <Box sx={{ ml: 25, pt: 8, display: 'flex', flexDirection: 'column' }}>
        <Grid container spacing={2}>
          <Grid item xs={12} md={6}>
            <Box sx={{ p: 3, bg: '#fff', borderRadius: 2, boxShadow: 3 }}>
              <h2 style={{margin:0}}>Temperature (°C)</h2>
              <h1 style={{fontSize: 48, margin:0}}>{env.temp ?? '-'}</h1>
            </Box>
          </Grid>
          <Grid item xs={12} md={6}>
            <Box sx={{ p: 3, bg: '#fff', borderRadius: 2, boxShadow: 3 }}>
              <h2 style={{margin:0}}>pH</h2>
              <h1 style={{fontSize: 48, margin:0}}>{env.pH ?? '-'}</h1>
            </Box>
          </Grid>
          {/* Add more widgets such as salinity, risk index… */}
        </Grid>
      </Box>
    </>
  );
}
